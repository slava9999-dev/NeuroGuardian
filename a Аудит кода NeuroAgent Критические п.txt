Диагноз
WB API v2/upload/task работает асинхронно. Успешный ответ 200 OK означает только то, что задача принята в очередь, но не то, что цены изменены.
Текущий код:

    // Обрабатывается только явная ошибка
  }

Проблема: Код считает операцию успешной сразу после получения Task ID, но не проверяет реальный статус выполнения задачи. Задача может упасть позже.
Исправление
typescriptDownloadCopy code// Добавить функцию проверки статуса задачи
export async function checkWbTaskStatus(
  apiKey: string,
  taskId: number
): Promise<{ completed: boolean; hasErrors: boolean; errors?: string[] }> {
  const response = await fetchWithRetry(
    `https://discounts-prices-api.wildberries.ru/api/v2/history/tasks`,
    {
      method: 'GET',
      headers: { Authorization: apiKey },
    }
  );

  if (!response.ok) {
    return { completed: false, hasErrors: true, errors: ['Failed to fetch task status'] };
  }

  const data = await response.json();
  const task = data.data?.find((t: any) => t.id === taskId);
  
  if (!task) {
    return { completed: false, hasErrors: false };
  }

  // Проверяем статус задачи и наличие ошибок в отдельных товарах
  const errors = task.details?.filter((d: any) => d.status === 'rejected' || d.errorText);
  
  return {
    completed: task.status === 'completed',
    hasErrors: errors?.length > 0,
    errors: errors?.map((e: any) => `nmID ${e.nmID}: ${e.errorText}`),
  };
}

export async function updateWbPrices(

  if (response.ok) {

    }
    
    const taskId = responseBody.data?.id;
    console.log(`📋 WB: Task created with ID: ${taskId}`);
    
    // Опционально: ждём и проверяем статус (с таймаутом)
    if (taskId) {

      const status = await checkWbTaskStatus(apiKey, taskId);

          taskId 
        };
      }
    }
    
    return { success: true, count: updates.length, taskId };
  }
}

typescriptDownloadCopy codeif (ozonResponse.ok) {
  const ozonData = await ozonResponse.json();
  const results = ozonData.result || [];
  
  const successfulUpdates = results.filter((r: any) => r.updated === true);
  const failedUpdates = results.filter((r: any) => r.updated === false || r.errors?.length > 0);
  
  if (failedUpdates.length > 0) {
    const errorMessages = failedUpdates
      .flatMap((f: any) => f.errors?.map((e: any) => `${f.product_id}: ${e.message}`) || [])

    ozonResult = { success: true, count: ozonUpdates.length, error: '' };
  }
  
  // Обновляем в БД только успешные
  for (const r of successfulUpdates) {
    const update = ozonUpdates.find(u => u.productId === `ozon-${r.product_id}`);
    if (update) {
