from sqlalchemy import Column, Integer, String, Text, DateTime, JSON
from sqlalchemy.ext.declarative import declarative_base
from openai import OpenAI
import json

Base = declarative_base()

class AgentMemory(Base):
    """Простая таблица памяти агента."""
    __tablename__ = "agent_memory"
    
    id = Column(Integer, primary_key=True)
    user_id = Column(String, index=True)  # tg_123456
    memory_type = Column(String)  # strategy, report, error, success
    content = Column(Text)  # Текст
    metadata = Column(JSON)  # {nm_ids: [123, 456], margin: 0.25}
    embedding = Column(String)  # Сериализованный вектор (для MVP просто текстовый поиск)
    created_at = Column(DateTime, default=datetime.utcnow)

class RAGLite:
    def __init__(self, db_session, openai_client: OpenAI):
        self.db = db_session
        self.openai = openai_client
    
    async def store(self, user_id: str, memory_type: str, content: str, metadata: dict = None):
        """Сохраняет событие в память."""
        
        # Для MVP: создаем эмбеддинг, но ищем просто по LIKE
        embedding = await self.openai.embeddings.create(
            input=content,
            model="text-embedding-3-small"
        )
        
        memory = AgentMemory(
            user_id=user_id,
            memory_type=memory_type,
            content=content,
            metadata=metadata or {},
            embedding=json.dumps(embedding.data[0].embedding)
        )
        
        self.db.add(memory)
        self.db.commit()
    
    async def search(self, user_id: str, query: str, limit: int = 5):
        """Ищет похожие события (для MVP - ILIKE, для продакшена - pgvector)."""
        
        # MVP: простой текстовый поиск
        results = self.db.query(AgentMemory).filter(
            AgentMemory.user_id == user_id,
            AgentMemory.content.ilike(f"%{query}%")
        ).order_by(AgentMemory.created_at.desc()).limit(limit).all()
        
        return [
            {"content": m.content, "metadata": m.metadata, "date": m.created_at}
            for m in results
        ]
    
    async def get_last_strategy(self, user_id: str):
        """Получает последнюю стратегию пользователя."""
        
        return self.db.query(AgentMemory).filter(
            AgentMemory.user_id == user_id,
            AgentMemory.memory_type == "strategy"
        ).order_by(AgentMemory.created_at.desc()).first()

from langchain_openai import ChatOpenAI
from langchain.memory import ConversationBufferMemory

class NeuroAgent:
    def __init__(self, rag: RAGLite, wb_tool, ozon_tool):
        self.rag = rag
        self.tools = [wb_tool, ozon_tool]
        self.llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.2)
        self.memory = ConversationBufferMemory()
    
    async def chat(self, user_id: str, message: str):
        # 1. Ищем контекст в памяти
        context = await self.rag.search(user_id, message, limit=3)
        
        # 2. Добавляем в промпт
        system_prompt = f"""
        Ты помощник продавца на WB/Ozon. 
        Контекст из прошлого: {context}
        Отвечай кратко, по-русски, с цифрами.
        """
        
        # 3. Определяем интент
        if "цена" in message:
            return await self.handle_price_request(user_id, message)
        elif "конкурент" in message:
            return await self.handle_competitor_request(user_id, message)
        
        # 4. Общий чат
        response = await self.llm.ainvoke([
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": message}
        ])
        
        # 5. Сохраняем в память
        await self.rag.store(user_id, "chat", message, {"response": response.content})
        
        return response.content

from pydantic import BaseModel, validator, conlist, Field
from datetime import datetime, timedelta
import asyncio

class WbPriceUpdate(BaseModel):
    """Валидация изменения цен WB."""
    
    nm_ids: conlist(int, min_items=1, max_items=500)
    change_type: str
    value: float = Field(ge=-50, le=50)  # Ограничение ±50%
    
    @validator('nm_ids')
    def validate_nm_ids(cls, v):
        if any(not isinstance(x, int) or x <= 0 for x in v):
            raise ValueError("Некорректный nm_id")
        return v
    
    @validator('change_type')
    def validate_change_type(cls, v):
        if v not in ["absolute", "percentage", "relative"]:
            raise ValueError("Неподдерживаемый тип изменения")
        return v

class SimpleCircuitBreaker:
    def __init__(self, threshold=3, timeout=30):
        self.threshold = threshold
        self.timeout = timeout
        self.failures = 0
        self.last_failure = None
        self._lock = asyncio.Lock()
    
    async def guard(self, func, *args, **kwargs):
        async with self._lock:
            now = datetime.now()
            
            # Проверяем, не открыт ли breaker
            if self.last_failure and (now - self.last_failure).seconds < self.timeout:
                if self.failures >= self.threshold:
                    raise Exception(f"WB API недоступен. Ожидание: {self.timeout - (now - self.last_failure).seconds}сек")
            
            try:
                result = await func(*args, **kwargs)
                self.failures = 0  # Сброс при успехе
                return result
            except Exception as e:
                self.failures += 1
                self.last_failure = now
                raise e

# Использование
circuit = SimpleCircuitBreaker()

async def update_prices_safe(wb_tool, data: dict):
    # Валидация
    validated = WbPriceUpdate(**data)
    
    # Выполнение с защитой
    return await circuit.guard(wb_tool.update_prices, validated)

queue = Queue("neuroagent", connection=redis_client)

def enqueue_price_update(user_id: str, operation_data: dict):
    """Добавляет задачу в очередь."""
    
    job = queue.enqueue(
        "workers.tasks.execute_price_update",
        user_id=user_id,
        operation_data=operation_data,
        job_timeout=300,  # 5 минут
        result_ttl=86400,  # Храним результат 1 день
        failure_ttl=604800  # Храним ошибку 7 дней
    )
    
    return job.id

        raise

from datetime import datetime

class AuditLog(Base):

    id = Column(Integer, primary_key=True)
    timestamp = Column(DateTime, default=datetime.utcnow, index=True)
    user_id = Column(String, index=True)

    error = Column(Text, nullable=True)
    ip_address = Column(String, nullable=True)
    llm_prompt = Column(Text, nullable=True)

def log_operation(db, user_id: str, operation: str, **kwargs):
    """Простой логгер, который пишет в Postgres."""
    
    log = AuditLog(user_id=user_id, operation=operation, **kwargs)
    db.add(log)
    db.commit()

class ConfirmationManager:
    def __init__(self, redis_client):
        self.redis = redis_client
    
    async def request_confirmation(self, user_id: str, operation: dict) -> str:
        """Генерирует код подтверждения."""
        
        confirm_id = str(uuid.uuid4())
        code = str(random.randint(100000, 999999))
        
        # Сохраняем в Redis на 10 минут
        await self.redis.setex(
            f"confirm:{confirm_id}",
            600,
            json.dumps({"user_id": user_id, "code": code, "operation": operation})
        )
        
        # Отправляем в Telegram
        await bot.send_message(
            user_id,
            f"🔐 Подтверждение операции:\n"
            f"Изменить цену на {len(operation['nm_ids'])} товаров?\n"
            f"Код: **{code}**\n"
            f"Отправьте код в ответ."
        )
        
        return confirm_id
    
    async def verify_code(self, confirm_id: str, user_code: str) -> bool:
        """Проверяет код."""
        
        data = await self.redis.get(f"confirm:{confirm_id}")
        if not data:
            return False
        
        data = json.loads(data)
        return data["code"] == user_code

@app.post("/api/chat")
async def chat_endpoint(
    request: ChatRequest,
    db: Session = Depends(get_db),
    user = Depends(verify_telegram_auth)
):
    # 1. Валидация входа
    validated_request = ChatRequest(**request.dict())
    
    # 2. Инициализация агента
    agent = NeuroAgent(
        rag=RAGLite(db, openai_client),
        wb_tool=WbPriceTool(user.wb_key),
        ozon_tool=OzonPriceTool(user.ozon_key)
    )
    
    # 3. Определяем интент
    if "цена" in validated_request.message and "подними" in validated_request.message:
        # 4. Парсим параметры
        brand = extract_brand(validated_request.message)  # "Nike"
        percent = extract_percent(validated_request.message)  # 15
        
        # 5. Ищем товары
        products = await agent.wb_tool.search_by_brand(brand)
        
        if len(products) > 10:
            # 6. Требуется подтверждение
            confirm_id = await confirmation_manager.request_confirmation(
                user.id,
                {
                    "nm_ids": [p.nm_id for p in products],
                    "change_type": "percentage",
                    "value": percent
                }
            )
            
            return {
                "status": "confirmation_required",
                "confirm_id": confirm_id,
                "message": f"Найдено {len(products)} товаров. Подтвердите кодом."
            }
        
        # 7. Выполняем сразу
        job_id = enqueue_price_update(user.id, {
            "nm_ids": [p.nm_id for p in products],
            "change_type": "percentage",
            "value": percent
        })
        
        return {
            "status": "queued",
            "job_id": job_id,
            "message": "Задача добавлена в очередь"
        }
    
    # 8. Обычный чат
    response = await agent.chat(user.id, validated_request.message)
    
    # 9. Логируем
    audit_logger.log(
        user_id=user.id,
        operation="chat",
        status="success",
        llm_prompt=validated_request.message
    )
    
    return {"status": "ok", "response": response}

function PriceUpdateStatus({ jobId }) {
  const [status, setStatus] = useState('pending');
  
  useEffect(() => {
    const interval = setInterval(async () => {
      const res = await fetch(`/api/job/${jobId}`);
      const data = await res.json();
      setStatus(data.status);
      
      if (data.status === 'completed') {
        clearInterval(interval);
        WebApp.showAlert(`Обновлено ${data.result.updated} товаров`);
      }
    }, 5000);
    
    return () => clearInterval(interval);
  }, [jobId]);
  
  return <ProgressBar status={status} />;
}
