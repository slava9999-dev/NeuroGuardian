  // Вспомогательные методы
  
  private isValidWBKeyFormat(key: string): boolean {
    // WB ключи обычно длинные строки без пробелов
    return key.length > 50 && !key.includes(' ') && /^[A-Za-z]/.test(key);
  }

  private async verifyWBConnection(apiKey: string): Promise<ConnectionResult> {
    try {
      const client = new WildberriesClient({ apiKey });
      const info = await client.getSellerInfo();
      const products = await client.getProducts();
      
      return {
        success: true,
        supplierId: info.supplierId,
        shopName: info.shopName,
        productsCount: products.length,
      };
    } catch (error) {
      return {
        success: false,
        error: this.parseApiError(error),
      };
    }
  }

  private async encryptApiKey(apiKey: string): Promise<string> {
    // Используем AES-256-GCM для шифрования
    const encryptionKey = process.env.API_KEY_ENCRYPTION_SECRET;
    return encrypt(apiKey, encryptionKey);
  }

export class UnitEconomicsCalculator {
  
  // Комиссии Wildberries по категориям (актуальные на 2025)
  private readonly WB_COMMISSIONS: Record<string, number> = {

    'Бытовая техника': 0.12,

    'Спорт': 0.15,
    'default': 0.15,

    'Одежда': 0.15,
    'Обувь': 0.15,
    'Электроника': 0.10,
    'Бытовая техника': 0.10,
    'Красота': 0.20,
    'Детские товары': 0.15,
    'Продукты': 0.08,
    'Товары для дома': 0.15,
    'default': 0.15,
  };

  // Скидка Ozon Card (оплачивается продавцом!)
  private readonly OZON_CARD_DISCOUNT_PERCENT = 0.05; // 5%

  // Логистика WB
  private readonly WB_LOGISTICS = {
    baseDeliveryCost: 55,        // Базовая доставка до ПВЗ
    perKgRate: 5,                // За каждый кг сверх 1
    perLiterRate: 3,             // За каждый литр сверх 1
    returnCostPercent: 0.33,     // 33% от стоимости доставки за возврат
    storagePerDayPerLiter: 0.07, // Хранение за литр в день
  };

  // Логистика Ozon

    firstMile: {

  /**
   * Полный расчёт юнит-экономики
   */
  async calculate(input: ProductCosts, options?: CalculationOptions): Promise<UnitEconomicsResult> {
    const warnings: EconomicsWarning[] = [];
    
    // 1. Комиссия маркетплейса
    const commissionRate = this.getCommissionRate(input.marketplace, input.category);
    const marketplaceCommission = input.sellingPrice * commissionRate;
    
    // 2. Эквайринг (обычно входит в комиссию, но нужно учитывать)
    const acquiringPercent = input.marketplace === 'ozon' ? 0.015 : 0.02;
    const acquiringFee = input.sellingPrice * acquiringPercent;
    
    // 3. Логистика
    const logisticsDetails = this.calculateLogistics(input);
    const logisticsCost = 
      logisticsDetails.firstMile + 
      logisticsDetails.lastMile + 
      logisticsDetails.storageCost;
    
    // 4. ⚠️ КРИТИЧНО: Ozon Card скидка (оплачивает продавец!)
    let ozonCardDiscount = 0;
    if (input.marketplace === 'ozon') {
      // По статистике ~40% покупок на Ozon с картой
      const ozonCardPurchaseRate = 0.40;
      ozonCardDiscount = input.sellingPrice * this.OZON_CARD_DISCOUNT_PERCENT * ozonCardPurchaseRate;
      
      if (ozonCardDiscount > input.sellingPrice * 0.02) {
        warnings.push({
          type: 'warning',
          code: 'OZON_CARD_IMPACT',
          message: `Скидка Ozon Card съедает ${(ozonCardDiscount / input.sellingPrice * 100).toFixed(1)}% от цены. Учтите это при ценообразовании!`,
          impact: ozonCardDiscount,
        });
      }
    }
    
    // 5. Возвраты
    const returnRate = options?.customReturnRate || this.getAverageReturnRate(input.category);
    const returnProcessingCost = (logisticsDetails.lastMile + logisticsDetails.returnCost) * returnRate;
    
    // 6. Упаковка
    const packagingCost = options?.packagingCost || 15; // Среднее значение
    
    // 7. Промо-акции
    const promoDiscount = options?.promoDiscount || 0;
    const promoCompensation = options?.promoCompensation || 0;
    
    // Итоговые расчёты
    const totalCosts = 
      input.costPrice +
      marketplaceCommission +
      acquiringFee +
      logisticsCost +
      ozonCardDiscount +
      returnProcessingCost +
      packagingCost +
      promoDiscount -
      promoCompensation;
    
    const grossProfit = input.sellingPrice - input.costPrice;
    const netProfit = input.sellingPrice - totalCosts;
    const marginPercent = (netProfit / input.sellingPrice) * 100;
    const roi = (netProfit / input.costPrice) * 100;
    
    // Минимальная безопасная цена
    const fixedCosts = packagingCost + logisticsDetails.firstMile;
    const variableCostRate = commissionRate + acquiringPercent + 
      (input.marketplace === 'ozon' ? this.OZON_CARD_DISCOUNT_PERCENT * 0.4 : 0);
    
    // Формула: Price = (costPrice + fixedCosts) / (1 - variableCostRate)
    const minSafePrice = (input.costPrice + fixedCosts) / (1 - variableCostRate);
    
    // Рекомендуемая цена с целевой маржой
    const targetMargin = options?.targetMarginPercent || 20;
    const recommendedMinPrice = (input.costPrice + fixedCosts) / (1 - variableCostRate - targetMargin / 100);
    
    // Проверка на критические проблемы
    if (marginPercent < 0) {
      warnings.push({
        type: 'critical',
        code: 'NEGATIVE_MARGIN',
        message: `КРИТИЧНО: Вы работаете в убыток! Маржа: ${marginPercent.toFixed(1)}%`,
        impact: netProfit,
      });
    } else if (marginPercent < 5) {
      warnings.push({
        type: 'critical',
        code: 'VERY_LOW_MARGIN',
        message: `Маржа критически низкая (${marginPercent.toFixed(1)}%). Любое изменение комиссий приведёт к убытку.`,
        impact: netProfit,
      });
    } else if (marginPercent < 10) {
      warnings.push({
        type: 'warning',
        code: 'LOW_MARGIN',
        message: `Маржа ниже рекомендуемой (${marginPercent.toFixed(1)}%). Рекомендуется минимум 15%.`,
        impact: (10 - marginPercent) / 100 * input.sellingPrice,
      });
    }

    return {
      sellingPrice: input.sellingPrice,
      costPrice: input.costPrice,
      
      marketplaceCommission,
      marketplaceCommissionPercent: commissionRate * 100,
      
      acquiringFee,
      acquiringPercent: acquiringPercent * 100,
      
      logisticsCost,
      logisticsDetails,
      
      ozonCardDiscount: input.marketplace === 'ozon' ? ozonCardDiscount : undefined,
      
      promoDiscount,
      promoCompensation,
      
      packagingCost,
      returnRate,
      returnProcessingCost,
      
      totalCosts,
      grossProfit,
      netProfit,
      marginPercent,
      roi,
      
      minSafePrice: Math.ceil(minSafePrice),
      recommendedMinPrice: Math.ceil(recommendedMinPrice),
      
      warnings,
    };
  }

  /**
   * Расчёт логистики
   */
  private calculateLogistics(input: ProductCosts): LogisticsDetails {
    if (input.marketplace === 'wildberries') {
      return this.calculateWBLogistics(input);
    } else {
      return this.calculateOzonLogistics(input);
    }
  }

  private calculateWBLogistics(input: ProductCosts): LogisticsDetails {
    const { weight, volume } = input;
    
    // First mile (до склада WB) - упрощённо
    const firstMile = 25; // Если FBS, иначе 0
    
    // Last mile (до покупателя)
    let lastMile = this.WB_LOGISTICS.baseDeliveryCost;
    if (weight > 1) {
      lastMile += (weight - 1) * this.WB_LOGISTICS.perKgRate;
    }
    if (volume > 1) {
      lastMile += (volume - 1) * this.WB_LOGISTICS.perLiterRate;
    }
    
    // Возврат
    const returnCost = lastMile * this.WB_LOGISTICS.returnCostPercent;
    
    // Хранение (предполагаем средний срок 14 дней)
    const averageStorageDays = 14;
    const storageCost = volume * this.WB_LOGISTICS.storagePerDayPerLiter * averageStorageDays;
    
    return { firstMile, lastMile, returnCost, storageCost };
  }

  private calculateOzonLogistics(input: ProductCosts): LogisticsDetails {
    const { weight } = input;
    
    // First mile
    let firstMile: number;
    if (weight <= 5) firstMile = this.OZON_LOGISTICS.firstMile.upTo5kg;
    else if (weight <= 10) firstMile = this.OZON_LOGISTICS.firstMile.upTo10kg;
    else if (weight <= 15) firstMile = this.OZON_LOGISTICS.firstMile.upTo15kg;
    else if (weight <= 25) firstMile = this.OZON_LOGISTICS.firstMile.upTo25kg;
    else firstMile = this.OZON_LOGISTICS.firstMile.over25kg;
    
    // Last mile
    const lastMile = this.OZON_LOGISTICS.lastMile.standard;
    
    // Возврат
    const returnCost = this.OZON_LOGISTICS.returnRate;
    
    // Хранение
    const averageStorageDays = 14;
    const storageCost = this.OZON_LOGISTICS.storagePerDayPerUnit * averageStorageDays;
    
    return { firstMile, lastMile, returnCost, storageCost };
  }

  private getCommissionRate(marketplace: string, category: string): number {
    const commissions = marketplace === 'wildberries' 
      ? this.WB_COMMISSIONS 
      : this.OZON_COMMISSIONS;
    
    return commissions[category] || commissions['default'];
  }

  private getAverageReturnRate(category: string): number {
    // Средние % возвратов по категориям
    const returnRates: Record<string, number> = {
      'Одежда': 0.25,      // 25% возвратов
      'Обувь': 0.30,       // 30% возвратов
      'Электроника': 0.05,
      'Красота': 0.08,
      'Детские товары': 0.15,
      'default': 0.10,
    };
    
    return returnRates[category] || returnRates['default'];
  }

  /**
   * Массовый расчёт для всех товаров пользователя
   */
  async calculateBulk(userId: string): Promise<BulkCalculationResult> {
    const products = await this.getProductsWithCosts(userId);
    const results: Map<string, UnitEconomicsResult> = new Map();
    
    let totalProfit = 0;
    let criticalCount = 0;
    let warningCount = 0;
    
    for (const product of products) {
      const result = await this.calculate(product);
      results.set(product.productId, result);
      
      totalProfit += result.netProfit;
      
      const criticalWarnings = result.warnings.filter(w => w.type === 'critical').length;
      const warningWarnings = result.warnings.filter(w => w.type === 'warning').length;
      
      if (criticalWarnings > 0) criticalCount++;
      if (warningWarnings > 0) warningCount++;
    }
    
    return {
      results,
      summary: {
        totalProducts: products.length,
        totalProfit,
        averageMargin: totalProfit / products.reduce((sum, p) => sum + p.sellingPrice, 0) * 100,
        criticalIssues: criticalCount,
        warnings: warningCount,
      },
    };
  }
}

    if (product.currentPrice < rule.minPrice) {
      threats.push({
        id: `below_min_${product.id}_${Date.now()}`,
        type: ThreatType.PRICE_BELOW_MINIMUM,
        severity: 'critical',
        
        productId: product.id,
        productName: product.name,
        marketplace: product.marketplace,
        
        description: `КРИТИЧНО: Текущая цена (${product.currentPrice}₽) ниже установленного минимума (${rule.minPrice}₽)!`,
        
        impact: {
          currentValue: product.currentPrice,
          threatenedValue: rule.minPrice,
          potentialLoss: rule.minPrice - product.currentPrice,
          potentialLossPercent: ((rule.minPrice - product.currentPrice) / rule.minPrice) * 100,
        },
        
        suggestedActions: [
          {
            type: 'adjust_price',
            description: `Немедленно поднять цену до минимума ${rule.minPrice}₽`,
            newPrice: rule.minPrice,
            confidence: 100,
          },
        ],
        
        detectedAt: new Date(),
        autoResolvable: true, // Это можно исправить автоматически
      });
    }
    
    // Маржа ниже целевой
    const economics = await this.economicsCalculator.calculate({
      productId: product.id,
      marketplace: product.marketplace,
      sellingPrice: product.currentPrice,
      costPrice: product.costPrice,
      weight: product.weight,
      volume: product.volume,
      category: product.category,
    });
    
    if (economics.marginPercent < rule.targetMargin) {
      const priceTotargetMargin = economics.recommendedMinPrice;
      
      threats.push({
        id: `low_margin_${product.id}_${Date.now()}`,
        type: ThreatType.MARGIN_BELOW_TARGET,
        severity: economics.marginPercent < 5 ? 'critical' : 'high',
        
        productId: product.id,
        productName: product.name,
        marketplace: product.marketplace,
        
        description: `Текущая маржа (${economics.marginPercent.toFixed(1)}%) ниже целевой (${rule.targetMargin}%)`,
        
        impact: {
          currentValue: economics.marginPercent,
          threatenedValue: rule.targetMargin,
          potentialLoss: (rule.targetMargin - economics.marginPercent) / 100 * product.currentPrice,
          potentialLossPercent: rule.targetMargin - economics.marginPercent,
        },
        
        suggestedActions: [
          {
            type: 'adjust_price',
            description: `Поднять цену до ${priceTotargetMargin}₽ для достижения целевой маржи`,
            newPrice: priceTotargetMargin,
            confidence: 90,
          },
          {
            type: 'request_approval',
            description: 'Запросить подтверждение на изменение цены',
            confidence: 100,
          },
        ],
        
        detectedAt: new Date(),
        autoResolvable: rule.autoAdjust,
      });
    }
    
    // Отрицательная маржа
    if (economics.marginPercent < 0) {
      threats.push({
        id: `negative_margin_${product.id}_${Date.now()}`,
        type: ThreatType.NEGATIVE_MARGIN,
        severity: 'critical',
        
        productId: product.id,
        productName: product.name,
        marketplace: product.marketplace,
        
        description: `УБЫТОК! Каждая продажа приносит убыток ${Math.abs(economics.netProfit).toFixed(0)}₽`,
        
        impact: {
          currentValue: economics.netProfit,
          threatenedValue: 0,
          potentialLoss: Math.abs(economics.netProfit),
          potentialLossPercent: Math.abs(economics.marginPercent),
        },
        
        suggestedActions: [
          {
            type: 'adjust_price',
            description: `СРОЧНО поднять цену до ${economics.minSafePrice}₽ (минимум для безубыточности)`,
            newPrice: economics.minSafePrice,
            confidence: 100,
          },
          {
            type: 'block_sales',
            description: 'Снять товар с продажи до пересмотра цены',
            confidence: 80,
          },
        ],
        
        detectedAt: new Date(),
        autoResolvable: true, // Критическая угроза - исправляем автоматически
      });
    }
    
    return threats;
  }

  private summarizeThreats(threats: Threat[]): ThreatSummary {
    return {
      total: threats.length,
      bySeverity: {
        critical: threats.filter(t => t.severity === 'critical').length,
        high: threats.filter(t => t.severity === 'high').length,
        medium: threats.filter(t => t.severity === 'medium').length,
        low: threats.filter(t => t.severity === 'low').length,
      },
      byType: threats.reduce((acc, t) => {
        acc[t.type] = (acc[t.type] || 0) + 1;
        return acc;
      }, {} as Record<ThreatType, number>),
      totalPotentialLoss: threats.reduce((sum, t) => sum + t.impact.potentialLoss, 0),
      autoResolvableCount: threats.filter(t => t.autoResolvable).length,
    };
  }
}

export interface PriceAdjustment {
  productId: string;
  marketplace: 'wildberries' | 'ozon';
  currentPrice: number;
  newPrice: number;
  reason: string;
  threatId?: string;
  requiresApproval: boolean;
}

export interface AdjustmentResult {
  adjustment: PriceAdjustment;
  success: boolean;
  error?: string;
  apiResponse?: any;
  executedAt: Date;
}

export class PriceShield {
  private wbClient: WildberriesClient;
  private ozonClient: OzonClient;
  private notificationService: NotificationService;
  private auditLog: AuditLogService;

  /**
   * Выполнить защиту цены по обнаруженной угрозе
   */
  async executeProtection(threat: Threat): Promise<AdjustmentResult | null> {
    // Находим подходящее автоматическое действие
    const autoAction = threat.suggestedActions.find(
      a => a.type === 'adjust_price' && a.confidence >= 85 && threat.autoResolvable
    );
    
    if (!autoAction || !autoAction.newPrice) {
      // Нет подходящего автоматического действия - отправляем алерт
      await this.sendAlertForManualAction(threat);
      return null;
    }
    
    const adjustment: PriceAdjustment = {
      productId: threat.productId,
      marketplace: threat.marketplace,
      currentPrice: threat.impact.currentValue,
      newPrice: autoAction.newPrice,
      reason: threat.description,
      threatId: threat.id,
      requiresApproval: false,
    };
    
    return this.executeAdjustment(adjustment);
  }

  /**
   * Выполнить корректировку цены
   */
  async executeAdjustment(adjustment: PriceAdjustment): Promise<AdjustmentResult> {
    const startTime = Date.now();
    
    // Логируем начало операции
    await this.auditLog.log({
      action: 'PRICE_ADJUSTMENT_STARTED',
      actor: 'price_shield',
      resource: adjustment.productId,
      details: adjustment,
    });
    
    try {
      let apiResponse: any;
      
      if (adjustment.marketplace === 'wildberries') {
        apiResponse = await this.adjustWBPrice(adjustment);
      } else {
        apiResponse = await this.adjustOzonPrice(adjustment);
      }
      
      const result: AdjustmentResult = {
        adjustment,
        success: true,
        apiResponse,
        executedAt: new Date(),
      };
      
      // Логируем успех
      await this.auditLog.log({
        action: 'PRICE_ADJUSTMENT_SUCCESS',
        actor: 'price_shield',
        resource: adjustment.productId,
        details: result,
        duration: Date.now() - startTime,
      });
      
      // Отправляем уведомление
      await this.notificationService.sendPriceAdjustmentNotification({
        ...adjustment,
        success: true,
      });
      
      return result;
      
    } catch (error) {
      const result: AdjustmentResult = {
        adjustment,
        success: false,
        error: error.message,
        executedAt: new Date(),
      };
      
      // Логируем ошибку
      await this.auditLog.log({
        action: 'PRICE_ADJUSTMENT_FAILED',
        actor: 'price_shield',
        resource: adjustment.productId,
        details: result,
        error: error.message,
        duration: Date.now() - startTime,
      });
      
      // Отправляем алерт об ошибке
      await this.notificationService.sendPriceAdjustmentNotification({
        ...adjustment,
        success: false,
        error: error.message,
      });
      
      return result;
    }
  }

  /**
   * Корректировка цены на Wildberries
   */
  private async adjustWBPrice(adjustment: PriceAdjustment): Promise<any> {
    // Валидация
    if (adjustment.newPrice < 1) {
      throw new Error('Цена не может быть меньше 1 рубля');
    }
    
    // WB API для обновления цен
    const response = await this.wbClient.updatePrice({
      nmId: parseInt(adjustment.productId.replace('wb_', '')),
      price: adjustment.newPrice,
    });
    
    // Проверяем результат
    if (response.error) {
      throw new Error(`WB API Error: ${response.errorText || response.error}`);
    }
    
    // Дополнительная проверка - убеждаемся, что цена применилась
    await this.sleep(2000); // Ждём 2 секунды
    
    const currentPrice = await this.wbClient.getProductPrice(
      parseInt(adjustment.productId.replace('wb_', ''))
    );
    
    if (Math.abs(currentPrice - adjustment.newPrice) > 1) {
      throw new Error(
        `Цена не применилась. Ожидали: ${adjustment.newPrice}, получили: ${currentPrice}`
      );
    }
    
    return response;
  }

  /**
   * Корректировка цены на Ozon
   */
  private async adjustOzonPrice(adjustment: PriceAdjustment): Promise<any> {
    // Валидация
    if (adjustment.newPrice < 1) {
      throw new Error('Цена не может быть меньше 1 рубля');
    }
    
    // Ozon API для обновления цен
    const response = await this.ozonClient.updatePrices([
      {
        product_id: parseInt(adjustment.productId.replace('ozon_', '')),
        price: adjustment.newPrice.toString(),
        // Устанавливаем old_price для отображения скидки если цена ниже
        old_price: adjustment.currentPrice > adjustment.newPrice 
          ? adjustment.currentPrice.toString() 
          : undefined,
      },
    ]);
    
    // Проверяем результат
    if (response.result?.length > 0) {
      const itemResult = response.result[0];
      if (!itemResult.updated) {
        throw new Error(
          `Ozon API Error: ${itemResult.errors?.join(', ') || 'Unknown error'}`
        );
      }
    }
    
    return response;
  }

  /**
   * Массовая корректировка цен
   */
  async executeBulkAdjustments(adjustments: PriceAdjustment[]): Promise<AdjustmentResult[]> {
    const results: AdjustmentResult[] = [];
    
    // Группируем по маркетплейсам для батч-обновления
    const wbAdjustments = adjustments.filter(a => a.marketplace === 'wildberries');
    const ozonAdjustments = adjustments.filter(a => a.marketplace === 'ozon');
    
    // WB позволяет обновлять до 1000 товаров за раз
    if (wbAdjustments.length > 0) {
      const wbBatches = this.chunk(wbAdjustments, 1000);
      for (const batch of wbBatches) {
        const batchResults = await this.executeBatchWB(batch);
        results.push(...batchResults);
      }
    }
    
    // Ozon позволяет обновлять до 1000 товаров за раз
    if (ozonAdjustments.length > 0) {
      const ozonBatches = this.chunk(ozonAdjustments, 1000);
      for (const batch of ozonBatches) {
        const batchResults = await this.executeBatchOzon(batch);
        results.push(...batchResults);
      }
    }
    
    return results;
  }

  private async executeBatchWB(adjustments: PriceAdjustment[]): Promise<AdjustmentResult[]> {
    try {
      const prices = adjustments.map(a => ({
        nmId: parseInt(a.productId.replace('wb_', '')),
        price: a.newPrice,
      }));
      
      const response = await this.wbClient.updatePrices(prices);
      
      // Обрабатываем результаты
      return adjustments.map((adjustment, index) => ({
        adjustment,
        success: !response.errors?.[index],
        error: response.errors?.[index],
        apiResponse: response,
        executedAt: new Date(),
      }));
      
    } catch (error) {
      // При ошибке batch-запроса - все считаем неудачными
      return adjustments.map(adjustment => ({
        adjustment,
        success: false,
        error: error.message,
        executedAt: new Date(),
      }));
    }
  }

  private async executeBatchOzon(adjustments: PriceAdjustment[]): Promise<AdjustmentResult[]> {
    try {
      const prices = adjustments.map(a => ({
        product_id: parseInt(a.productId.replace('ozon_', '')),
        price: a.newPrice.toString(),
      }));
      
      const response = await this.ozonClient.updatePrices(prices);
      
      // Обрабатываем результаты
      return adjustments.map((adjustment, index) => {
        const itemResult = response.result?.[index];
        return {
          adjustment,
          success: itemResult?.updated ?? false,
          error: itemResult?.errors?.join(', '),
          apiResponse: response,
          executedAt: new Date(),
        };
      });
      
    } catch (error) {
      return adjustments.map(adjustment => ({
        adjustment,
        success: false,
        error: error.message,
        executedAt: new Date(),
      }));
    }
  }

  private async sendAlertForManualAction(threat: Threat): Promise<void> {
    await this.notificationService.sendThreatAlert(threat);
  }

  private chunk<T>(array: T[], size: number): T[][] {
    const chunks: T[][] = [];
    for (let i = 0; i < array.length; i += size) {
      chunks.push(array.slice(i, i + size));
    }
    return chunks;
  }

  private sleep(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

export interface MonitoringResult {
  runId: string;
  startedAt: Date;
  completedAt: Date;
  duration: number;
  
  // Статистика
  stats: {
    usersProcessed: number;
    productsChecked: number;
    threatsDetected: number;
    autoAdjustments: number;
    alertsSent: number;
    errors: number;
  };
  
  // Детали по пользователям
  userResults: UserMonitoringResult[];
}

export interface UserMonitoringResult {
  userId: string;
  productsChecked: number;
  threats: Threat[];
  adjustments: AdjustmentResult[];
  notificationSent: boolean;
}

export class SentinelService {
  private threatDetector: ThreatDetector;
  private priceShield: PriceShield;
  private notificationService: NotificationService;
  private db: Database;

  /**
   * Запуск полного цикла мониторинга
   */
  async runFullMonitoringCycle(): Promise<MonitoringResult> {
    const runId = `monitoring_${Date.now()}`;
    const startedAt = new Date();
    
    console.log(`[Sentinel] Starting monitoring cycle ${runId}`);
    
    const stats = {
      usersProcessed: 0,
      productsChecked: 0,
      threatsDetected: 0,
      autoAdjustments: 0,
      alertsSent: 0,
      errors: 0,
    };
    
    const userResults: UserMonitoringResult[] = [];
    
    try {
      // Получаем всех активных пользователей с настроенными правилами
      const activeUsers = await this.getActiveUsers();
      
      for (const user of activeUsers) {
        try {
          const userResult = await this.processUser(user);
          userResults.push(userResult);
          
          stats.usersProcessed++;
          stats.productsChecked += userResult.productsChecked;
          stats.threatsDetected += userResult.threats.length;
          stats.autoAdjustments += userResult.adjustments.filter(a => a.success).length;
          if (userResult.notificationSent) stats.alertsSent++;
          
        } catch (error) {
          console.error(`[Sentinel] Error processing user ${user.id}:`, error);
          stats.errors++;
        }
      }
      
    } catch (error) {
      console.error(`[Sentinel] Critical error in monitoring cycle:`, error);
      stats.errors++;
    }
    
    const completedAt = new Date();
    const duration = completedAt.getTime() - startedAt.getTime();
    
    const result: MonitoringResult = {
      runId,
      startedAt,
      completedAt,
      duration,
      stats,
      userResults,
    };
    
    // Сохраняем результат в БД
    await this.saveMonitoringResult(result);
    
    // Логируем итоги
    console.log(`[Sentinel] Cycle ${runId} completed in ${duration}ms:`, stats);
    
    return result;
  }

  /**
   * Обработка одного пользователя
   */
  private async processUser(user: User): Promise<UserMonitoringResult> {
    // 1. Синхронизируем актуальные цены с маркетплейсов
    await this.syncCurrentPrices(user.id);
    
    // 2. Сканируем угрозы
    const scanResult = await this.threatDetector.scanAllThreats(user.id);
    
    // 3. Обрабатываем угрозы
    const adjustments: AdjustmentResult[] = [];
    const criticalThreats: Threat[] = [];
    
    for (const threat of scanResult.threats) {
      if (threat.autoResolvable && threat.severity === 'critical') {
        // Критические авто-разрешимые угрозы - исправляем немедленно
        const result = await this.priceShield.executeProtection(threat);
        if (result) {
          adjustments.push(result);
        }
      } else if (threat.severity === 'critical' || threat.severity === 'high') {
        // Не авто-разрешимые критические - собираем для алерта
        criticalThreats.push(threat);
      }
    }
    
    // 4. Отправляем уведомление пользователю
    let notificationSent = false;
    
    if (criticalThreats.length > 0 || adjustments.length > 0) {
      await this.sendUserNotification(user, {
        threats: criticalThreats,
        adjustments,
        totalChecked: scanResult.totalProducts,
      });
      notificationSent = true;
    }
    
    return {
      userId: user.id,
      productsChecked: scanResult.totalProducts,
      threats: scanResult.threats,
      adjustments,
      notificationSent,
    };
  }

  /**
   * Синхронизация актуальных цен
   */
  private async syncCurrentPrices(userId: string): Promise<void> {
    const connections = await this.getUserMarketplaceConnections(userId);
    
    for (const connection of connections) {
      try {
        if (connection.marketplace === 'wildberries') {
          const prices = await this.wbClient.getPrices(connection.apiKey);
          await this.updateProductPrices(userId, 'wildberries', prices);
        } else if (connection.marketplace === 'ozon') {
          const prices = await this.ozonClient.getPrices(connection.clientId, connection.apiKey);
          await this.updateProductPrices(userId, 'ozon', prices);
        }
      } catch (error) {
        console.error(`[Sentinel] Failed to sync prices for ${connection.marketplace}:`, error);
      }
    }
  }

  /**
   * Отправка уведомления пользователю
   */
  private async sendUserNotification(
    user: User, 
    data: { threats: Threat[], adjustments: AdjustmentResult[], totalChecked: number }
  ): Promise<void> {
    const message = this.formatMonitoringNotification(data);
    
    await this.notificationService.sendTelegram(user.telegramChatId, message);
  }

  /**
   * Форматирование уведомления
   */
  private formatMonitoringNotification(data: {
    threats: Threat[],
    adjustments: AdjustmentResult[],
    totalChecked: number,
  }): string {
    const lines: string[] = [];
    
    lines.push(`🛡️ *Отчёт мониторинга*`);
    lines.push(`⏰ ${new Date().toLocaleString('ru-RU')}`);
    lines.push(``);
    lines.push(`📦 Проверено товаров: ${data.totalChecked}`);
    
    // Автоматические корректировки
    if (data.adjustments.length > 0) {
      lines.push(``);
      lines.push(`✅ *Автоматически скорректировано: ${data.adjustments.length}*`);
      
      for (const adj of data.adjustments.slice(0, 5)) {
        if (adj.success) {
          lines.push(`  • ${adj.adjustment.productId}: ${adj.adjustment.currentPrice}₽ → ${adj.adjustment.newPrice}₽`);
        }
      }
      
      if (data.adjustments.length > 5) {
        lines.push(`  ...и ещё ${data.adjustments.length - 5}`);
      }
    }
    
    // Угрозы требующие внимания
    if (data.threats.length > 0) {
      lines.push(``);
      lines.push(`⚠️ *Требуют внимания: ${data.threats.length}*`);
      
      const criticalCount = data.threats.filter(t => t.severity === 'critical').length;
      const highCount = data.threats.filter(t => t.severity === 'high').length;
      
      if (criticalCount > 0) {
        lines.push(`  🚨 Критических: ${criticalCount}`);
      }
      if (highCount > 0) {
        lines.push(`  🔶 Важных: ${highCount}`);
      }
      
      // Показываем первые 3 угрозы
      for (const threat of data.threats.slice(0, 3)) {
        const icon = threat.severity === 'critical' ? '🚨' : '⚠️';
        lines.push(`  ${icon} ${threat.description.slice(0, 60)}...`);
      }
      
      if (data.threats.length > 3) {
        lines.push(``);
        lines.push(`📊 Подробнее в личном кабинете`);
      }
    }
    
    if (data.threats.length === 0 && data.adjustments.length === 0) {
      lines.push(``);
      lines.push(`✅ Все цены в безопасности!`);
    }
    
    return lines.join('\n');
  }

  private async getActiveUsers(): Promise<User[]> {
    const result = await this.db.query(`
      SELECT u.* FROM users u
      JOIN marketplace_connections mc ON u.id = mc.user_id
      WHERE u.monitoring_enabled = true
        AND mc.is_active = true
      GROUP BY u.id
    `);
    return result.rows;
  }

  private async saveMonitoringResult(result: MonitoringResult): Promise<void> {
    await this.db.query(`
      INSERT INTO monitoring_runs (run_id, started_at, completed_at, duration, stats, user_count)
      VALUES ($1, $2, $3, $4, $5, $6)
    `, [
      result.runId,
      result.startedAt,
      result.completedAt,
      result.duration,
      JSON.stringify(result.stats),
      result.userResults.length,
    ]);
  }
}

export class AgentKnowledgeBase {

  private readonly DOCS_PATH = './knowledge/';

  /**
   * Загрузить все документы базы знаний
   */
  async initialize(): Promise<void> {

  /**

   */
  async search(query: string, options?: SearchOptions): Promise<SearchResult[]> {
    const normalizedQuery = this.normalizeQuery(query);
    const results: SearchResult[] = [];
    
    for (const doc of this.documents.values()) {
      // Фильтр по источнику
      if (options?.sources && !options.sources.includes(doc.source)) {
        continue;
      }
      
      // Вычисляем релевантность
      const score = this.calculateRelevance(normalizedQuery, doc);
      
      if (score > 0.3) { // Минимальный порог релевантности
        results.push({
          document: doc,
          score,
          snippet: this.extractSnippet(doc.content, normalizedQuery),
        });
      }
    }
    
    // Сортируем по релевантности
    results.sort((a, b) => b.score - a.score);
    
    // Ограничиваем количество результатов
    return results.slice(0, options?.limit || 5);
  }

  /**
   * Получить документ для контекста ответа агента
   */
  async getContextForQuestion(question: string): Promise<AgentContext> {
    // Определяем тему вопроса
    const topic = this.identifyTopic(question);
    
    // Ищем релевантные документы
    const relevantDocs = await this.search(question, {
      limit: 3,
      sources: this.getSourcesForTopic(topic),
    });
    
    // Формируем контекст
    return {
      topic,
      relevantDocuments: relevantDocs,
      systemPromptAddition: this.buildSystemPromptAddition(relevantDocs),
    };
  }

  /**
   * Проверка актуальности документации
   */
  async verifyDocumentation(): Promise<VerificationResult> {
    const result: VerificationResult = {
      totalDocuments: this.documents.size,
      bySource: {},
      outdated: [],
      missing: [],
      lastVerified: new Date(),
    };
    
    // Считаем по источникам
    for (const doc of this.documents.values()) {
      result.bySource[doc.source] = (result.bySource[doc.source] || 0) + 1;
      
      // Проверяем актуальность (документы старше 90 дней)
      const daysSinceUpdate = (Date.now() - doc.lastUpdated.getTime()) / (1000 * 60 * 60 * 24);
      if (daysSinceUpdate > 90) {
        result.outdated.push({
          id: doc.id,
          title: doc.title,
          lastUpdated: doc.lastUpdated,
          daysOld: Math.floor(daysSinceUpdate),
        });
      }
    }
    
    // Проверяем наличие обязательных тем
    const requiredTopics = [
      // Wildberries
      'wb_api_prices',
      'wb_api_products', 
      'wb_commissions',
      'wb_logistics_tariffs',
      'wb_fbs_fbo',
      
      // Ozon
      'ozon_api_prices',
      'ozon_api_products',
      'ozon_commissions',
      'ozon_card_rules',
      'ozon_logistics',
      'ozon_premium',
      
      // Общее
      'unit_economics',
      'price_protection_rules',
    ];
    
    for (const topic of requiredTopics) {
      const found = Array.from(this.documents.values()).some(
        doc => doc.tags.includes(topic)
      );
      if (!found) {
        result.missing.push(topic);
      }
    }
    
    return result;
  }

  private calculateRelevance(query: string, doc: KnowledgeDocument): number {
    const queryTerms = query.toLowerCase().split(/\s+/);
    const titleLower = doc.title.toLowerCase();
    const contentLower = doc.content.toLowerCase();
    const tags = doc.tags.map(t => t.toLowerCase());
    
    let score = 0;
    
    for (const term of queryTerms) {
      // Точное совпадение в тегах - высший балл
      if (tags.includes(term)) score += 0.4;
      
      // Совпадение в заголовке
      if (titleLower.includes(term)) score += 0.3;
      
      // Совпадение в контенте
      const contentMatches = (contentLower.match(new RegExp(term, 'g')) || []).length;
      score += Math.min(contentMatches * 0.05, 0.2);
    }
    
    return Math.min(score, 1);
  }

  private extractSnippet(content: string, query: string): string {
    const terms = query.split(/\s+/);
    const firstTerm = terms[0];
    
    const index = content.toLowerCase().indexOf(firstTerm.toLowerCase());
    if (index === -1) return content.slice(0, 200) + '...';
    
    const start = Math.max(0, index - 100);
    const end = Math.min(content.length, index + 200);
    
    let snippet = content.slice(start, end);
    if (start > 0) snippet = '...' + snippet;
    if (end < content.length) snippet = snippet + '...';
    
    return snippet;
  }

  private buildSystemPromptAddition(docs: SearchResult[]): string {
    if (docs.length === 0) return '';
    
    let addition = '\n\n--- Релевантная информация из базы знаний ---\n\n';
    
    for (const doc of docs) {
      addition += `### ${doc.document.title}\n`;
      addition += doc.snippet + '\n\n';
    }
    
    addition += '--- Конец информации из базы знаний ---\n';
    
    return addition;
  }
}

    // 6. Проверка audit logging
    checks.push(...await this.checkAuditLogging());
    
    // 7. Проверка production safety
    checks.push(...await this.checkProductionSafety());
    
    // Подсчёт результатов
    const passed = checks.filter(c => c.status === 'passed');
    const failed = checks.filter(c => c.status === 'failed');
    const warnings = checks.filter(c => c.status === 'warning');
    
    // Расчёт оценки
    const criticalFailed = failed.filter(f => f.severity === 'critical').length;
    const highFailed = failed.filter(f => f.severity === 'high').length;
    
    let score = 100;
    score -= criticalFailed * 25;
    score -= highFailed * 15;
    score -= failed.filter(f => f.severity === 'medium').length * 5;
    score -= warnings.length * 2;
    score = Math.max(0, score);
    
    return {
      overallScore: score,
      passed,
      failed,
      warnings,
      recommendations: this.generateRecommendations(failed, warnings),
    };
  }

  private parseExpiry(expiry: string): number {
    const match = expiry.match(/^(\d+)([dhms])$/);
    if (!match) return 1;
    
    const value = parseInt(match[1]);
    const unit = match[2];
    
    switch (unit) {
      case 'd': return value;
      case 'h': return value / 24;
      case 'm': return value / 1440;
      case 's': return value / 86400;
      default: return 1;
    }

3. Защита цены учитывает все критические угрозы цены и юнит‑экономики, включая скидку по “Ozon карте”/программам, где скидку оплачивает продавец (т.е. влияет на фактическую цену/маржу селлера).
4. Уведомления в Telegram приходят селлеру:

регулярные отчеты о мониторинге (каждые 30 минут, с троттлингом/объединением),
мгновенные алерты по критическим событиям (падение маржи, “цена ниже минимума”, ошибки API, рассинхрон).


5. Агент в чате отвечает, опираясь на актуальную документацию Ozon/WB из вашей базы знаний (RAG), с цитированием источника/версии, и не “галлюцинирует” при отсутствии данных.
6. Все 6 автоматизаций n8n проверены end‑to‑end, задокументированы, имеют ретраи/идемпотентность/алерты и не ломают production.
7. Проведена security hardening: секреты, доступы, аудит, запреты опасных режимов, защита от ошибочных массовых изменения цен.


2) Сценарии (User Stories)
2.1 Онбординг нового селлера (обязательный)
Как селлер, я впервые пишу агенту → агент:

1. объясняет шаги подключения,
2. просит подключить WB/Ozon (ключи/Client‑Id/Api‑Key) через безопасный UI (не в чат),
3. валидирует доступы (read + price update scopes),
4. делает первичную синхронизацию товаров,
5. предлагает настроить правила защиты цены (по товарам/группам),
6. предлагает заполнить юнит‑экономику (комиссии/логистика/себестоимость/скидки/эквайринг/возвраты и пр.),
7. только после состояния READY включает мониторинг/авто‑изменение цен.

Критерий приемки: пока магазин не READY, агент не делает авто‑действий и не “рассчитывает” на пустых данных; любые действия — только в режиме подсказок.

2.2 Защита цены (“сторож цены”)
Как селлер, я задаю число “защищаемая цена” (и/или минимальная маржа) → система:

* вычисляет, какая “цена в кабинете/базовая цена” должна быть выставлена на WB/Ozon, чтобы:

выдержать защищаемую цену/маржу с учетом комиссий/логистики/скидок,
учесть особые механики скидок, в т.ч. скидку, оплачиваемую селлером (Ozon Card/подобные программы),
не уйти в минус и не нарушить ограничения (min/max, лимит изменения, “заморозка” при аномалиях).


* каждые 30 минут проверяет и при отклонении — автоматически выравнивает цену через API (если разрешено правилом), иначе шлет алерт.

Критерий приемки: изменение цены в МП подтверждается реальным API ответом, логируется в audit, отображается в дашборде и отражается в следующем цикле мониторинга.

2.3 Проактивная помощь
Как селлер, я хочу, чтобы агент сам подсказывал:

* какие данные не заполнены (себестоимость/логистика/процент выкупа/программы скидок),
* где риск “продажа в минус”,
* что мешает точному расчету (рассинхрон, промо‑акции, ограничения API),
* какие товары требуют правил.

Критерий приемки: агент имеет список “missing critical inputs” и формирует план действий пользователю.

3) Функциональные требования
3.1 Интеграции Wildberries и Ozon (Real‑time readiness)
Требования:

* Раздельные коннекторы WB и Ozon с:

rate‑limit, backoff, ретраи,
нормализацией ошибок (401/403/429/5xx),
идемпотентностью (особенно при повторных запусках n8n),
журналированием запросов без утечки секретов.


* Поддержка операций минимум:

получить список товаров/sku/offer_id,
получить текущие цены/остатки,
установить/обновить цены,
(по возможности) получить данные о скидках/программах, влияющих на цену селлера.



Критерий приемки: на “лайф” ключах селлера выполняются read‑операции и тестовая update‑операция на тестовом товаре (или в dry‑run, если прод‑риски).

3.2 Синхронизация товаров
Требования:

* Первичная полная синхронизация (pull) + последующая инкрементальная (по расписанию/по изменившимся).
* Маппинг сущностей:

WB: nmId/артикул/баркод…
Ozon: product_id/offer_id/sku…


* Хранение истории синхронизации (когда, что обновилось, сколько позиций, ошибки).

Критерий приемки: после sync в БД есть консистентный каталог товаров, на который можно навесить правила защиты цены.

3.3 Калькулятор расчета цены и юнит‑экономика (обязательно)
Задача: проверить и довести “логику калькулятора” до уровня production.
Требования к модели расчета (минимум):

* входные параметры по товару/каналу:

себестоимость (COGS),
логистика (до клиента/фулфилмент/последняя миля — как модель проекта предполагает),
комиссия МП,
эквайринг/платежные комиссии,
налоги (если предусмотрено настройками),
хранение/платные услуги/обработка возвратов (как минимум коэффициентом),
рекламные расходы (опционально, но закладка должна быть),
скидки, оплачиваемые продавцом (ключевой кейс: Ozon Card и аналоги) — как отдельная статья, влияющая на net‑revenue.


* выход:

защищаемая величина (настраиваемо):
A) минимальная маржа, или
B) минимальная net‑прибыль, или
C) целевая “финальная цена покупателю”, или
D) комбинированно (например: “цена покупателю ≤ X, но маржа ≥ Y”).
рекомендуемая цена, которую нужно выставить через API (base/list price).


* “Explainability”: агент/дашборд показывают разложение “куда делись деньги” (понятным текстом и цифрами).

Критерии приемки:

* Есть набор unit‑тестов калькулятора (минимум 20 кейсов), включая:

обычная продажа,
продажа со скидкой, оплачиваемой селлером,
пограничные случаи (нулевые/отрицательные значения, округления),
рост комиссии/логистики,
запрет “ухода в минус”.


* Расчет воспроизводим: одинаковые входные → одинаковый результат.


3.4 “Сторож цены” — правила защиты и автокоррекция цен
Требования:

* Правила на уровень: товар / группа / магазин / маркетплейс.
* В правилах минимум:

защищаемая цена (что именно защищаем — настраиваемо),
min/max границы,
лимит шага изменения (например, не более N% за цикл),
режим: auto‑apply или “только уведомлять”,
поведение при промо/акциях (например, “не трогать, но предупреждать”),
поведение при недоступности API (fallback в уведомления).


* Детектор угроз:

“цена ниже минимума”,
“маржа ниже порога”,
“скидка продавца активна → риск убытка”,
“аномальное изменение конкурентов/рынка”,
“рассинхрон: в БД одно, в МП другое”.



Критерий приемки: в каждом цикле мониторинга система либо:

* обновляет цену (и подтверждает), либо
* шлет алерт с четкой причиной и рекомендованной ценой, либо
* ставит “safety stop” при аномалии (и эскалирует админу).


3.5 Анализ конкурентов
Требования:

* Сбор конкурентных цен по заданным правилам (список конкурентов/категория/топ‑N).
* Стратегия корректировки: “держать X% выше/ниже”, но никогда не ниже min‑маржи.
* Логи и репорты: что увидели, что сделали, почему.

Критерий приемки: конкурентный модуль не может заставить цену уйти в минус; любые рекомендации ограничены юнит‑экономикой.

3.6 Агент и база знаний: актуальная документация WB/Ozon
Требования (RAG/Knowledge QA):

* Документация, которую вы подгружали, должна:

иметь метаданные: источник URL/раздел/дата обновления/версия,
поддерживать “freshness policy” (например: помечать как устаревшую после N дней).


* В ответах агент обязан:

ссылаться на конкретный фрагмент/источник (“цитата‑ссылка/идентификатор документа”),
при отсутствии уверенности говорить “в документации в контексте не найдено” и предлагать шаги проверки, а не выдумывать.


* Набор контрольных вопросов (knowledge tests) по ключевым endpoint’ам и ограничениям API.

Критерий приемки: прогоняется тест‑набор вопросов; точность/полнота в рамках вашей базы знаний; ответы содержат ссылку на источник.

4) n8n: проверка 6 автоматизаций (обязательная)
Требование: каждая автоматизация описана паспортом и проходит E2E тест.
Паспорт автоматизации (для каждой из 6)

1. Название / ID
2. Триггер (cron/вебхук/событие)
3. Входы (какие данные/секреты нужны)
4. Шаги (ноды, что делают)
5. Выходы (БД записи, API вызовы, TG)
6. Идемпотентность (как избегаем дублей)
7. Ретраи и DLQ (как обрабатываем падения)
8. Алёрты (когда и кому)
9. SLA (что считается “успешно выполнено”)
10. Тест‑сценарий (как руками/автотестом проверить)

Минимальный набор логически необходимых автоматизаций (если ваши 6 совпадают — отлично)

1. Sync товаров WB
2. Sync товаров Ozon
3. Мониторинг/сторож цены (каждые 30 минут)
4. Отчет в TG (каждые 30 минут/по итогам мониторинга)
5. Healthcheck/Watchdog (API, БД, n8n, очереди)
6. Ошибки/инциденты (агрегация, дедуп, эскалация)

Критерий приемки: каждая из 6 выполняется в staging/production‑like окружении, с логами, ретраями, и не вызывает опасных побочных эффектов при повторном запуске.

5) Telegram уведомления (каждые 30 минут + критические)
Требования:

* Каналы: селлер (личные уведомления) + админ (инциденты).
* Типы сообщений:

“Мониторинг выполнен”: сколько товаров проверено, сколько изменено, топ‑риски.
“Цена/маржа под угрозой”: товар, текущая, рекомендованная, причина, что делать.
“Ошибка API/лимиты/403”: с инструкцией.


* Троттлинг/дедуп:

объединять события в один отчет,
не спамить одинаковыми алертами (cooldown по товару/проблеме).



Критерий приемки: сообщения приходят стабильно; при массовых изменениях не превращаются в 200 уведомлений подряд.

6) Дашборды и “лайф режим”
Требования к дашбордам:

* Статус коннекторов (WB/Ozon): последний успех/ошибка/время ответа.
* Статус мониторинга: последний цикл, следующий цикл, длительность, % ошибок.
* Таблица “сторож цены”: текущая цена, защищаемая цель, отклонение, последнее действие.
* Юнит‑экономика по товарам: маржа, риск, влияние скидок (в т.ч. селлер‑funded).
* Журнал действий (audit trail): кто/что/когда изменил цену, причина, request id.

Критерий приемки: после запуска в лайф режиме дашборд отражает реальные действия и состояния (не “примерные”).

7) Безопасность (приоритет №1)
7.1 Секреты и доступы

* API ключи WB/Ozon хранятся зашифрованно (KMS/Secret Manager/DB encryption), никогда не логируются.
* Разделение окружений: dev/staging/prod; запрет использовать prod‑секреты в dev.
* RBAC: селлер видит только свои магазины/товары/логи.
* “Dangerous mode” запрещен в prod: любые тестовые мок‑режимы, демо‑аккаунты, bypass‑флаги.

7.2 Safety‑ограничители для цен

* Лимит изменения цены за цикл и за сутки.
* “Kill switch” на магазин/товар/пользователя.
* Обязательный dry‑run режим для первого включения авто‑apply (например, 1–2 цикла “только уведомлять”).
* Блокировка изменений при:

невалидных входных данных юнит‑экономики,
обнаружении промо/акции, где расчет не уверен,
резком скачке параметров (аномалия/возможная ошибка API).



7.3 Аудит и расследования

* Иммутабельный аудит: запись каждого изменения цены, входных данных, версии алгоритма, источника решения.
* Correlation ID для каждого цикла мониторинга и каждого API вызова.

Критерий приемки: можно ответить на вопрос “почему цена изменилась” за 1 минуту, открыв audit запись.

8) Тестирование и приемка (обязательный набор)
8.1 Unit tests

* Калькулятор (включая скидки продавца/“Ozon карта”).
* Правила защиты (min/max/лимиты/stop‑conditions).
* Нормализация данных (WB/Ozon → unified model).

8.2 Integration tests

* Подключение WB/Ozon (read‑операции обязательно; update — controlled).
* Проверка n8n вебхуков (auth, payload, ретраи).
* Telegram отправка (в тестовый чат).

8.3 E2E тест “лайф”
Сценарий:

1. Новый пользователь → онбординг → подключение → sync
2. Настройка защищаемой цены на 2–3 товара
3. Прогон мониторинга (30 минут можно имитировать ручным запуском)
4. Система корректирует цену через API или шлет алерт
5. Отчет приходит в TG, запись есть в audit, дашборд обновлен

Критерий приемки: все шаги выполняются без ручных “подкруток в базе”.

9) Технические deliverables (что должен сделать агент‑кодер)

1. Документированный Onboarding state machine (NEW → CONNECTING → SYNCING → CONFIGURING → READY).
2. Исправления/доработки коннекторов WB/Ozon под update‑операции цен + ограничения.
3. Проверка/исправление калькулятора и юнит‑тесты.
4. Реализация/проверка “сторожа цены” с учетом seller‑funded скидок.
5. Валидация RAG базы знаний + тест‑набор вопросов по докам.
6. Полная проверка 6 n8n автоматизаций по “паспортам”.
7. Telegram уведомления (30 минут + критика) с дедупом.
8. Дашборды статуса/аудита/рисков.
9. Security hardening: секреты, RBAC, запреты опасных режимов, аудит, kill‑switch, лимиты изменений.
10. Runbook: “как подключить магазин”, “что делать при 403/429”, “как откатить изменения цены”.


10) Приложение: минимальные правила вычисления “выравнивания” цены (формализация)
Чтобы избежать разночтений, в системе должно быть явно задано что именно защищаем (настройка на товар/профиль):

* Вариант A (самый частый): защищаем минимальную маржу/прибыль селлера. Тогда “выравнивание” поднимает цену, если из‑за скидок (включая seller‑funded) маржа падает ниже порога.
* Вариант B: защищаем финальную цену покупателю (например, “покупатель должен видеть не ниже X”). Тогда система пересчитывает базовую цену так, чтобы с учетом всех скидок итоговая была X (но при этом проверяет маржу и может запретить выполнение, если это убыточно).
* Вариант C: гибрид: “финальная цена не ниже X” и “маржа не ниже Y”.

Критерий приемки: выбранный режим отображается в правилах и объясняется в уведомлениях (“что защищаем и почему изменили цену”).

Если вы передадите это ТЗ агенту‑кодеру как есть, он сможет последовательно: (1) зафиксировать логику защиты/расчета, (2) довести интеграции и мониторинг до лайф‑готовности, (3) прогнать n8n/телеграм/дашборды end‑to‑end, (4) закрыть безопасность так, чтобы “авто‑прайсинг” не превратился в “авто‑катастрофу”.I prefer this
