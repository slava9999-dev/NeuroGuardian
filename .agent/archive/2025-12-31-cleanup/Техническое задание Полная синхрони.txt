services:
  # ============================================
  # PostgreSQL - Основная база данных
  # ============================================
  postgres:
    image: postgres:16-alpine
    container_name: ng_postgres
    restart: unless-stopped
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-neuroguardian}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-localdevpassword}
      POSTGRES_DB: ${POSTGRES_DB:-neuroguardian_dev}
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./postgres/init.sql:/docker-entrypoint-initdb.d/init.sql:ro
    ports:
      - "5432:5432"
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-neuroguardian}"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - ng_network

  # ============================================
  # Redis - Кэширование и очереди
  # ============================================
  redis:
    image: redis:7-alpine
    container_name: ng_redis
    restart: unless-stopped
    command: redis-server --appendonly yes --requirepass ${REDIS_PASSWORD:-localredispass}
    volumes:
      - redis_data:/data
    ports:
      - "6379:6379"
    healthcheck:
      test: ["CMD", "redis-cli", "-a", "${REDIS_PASSWORD:-localredispass}", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - ng_network

  # ============================================
  # n8n - Автоматизация и workflow
  # ============================================
  n8n:
    image: n8nio/n8n:latest
    container_name: ng_n8n
    restart: unless-stopped
    environment:
      - N8N_BASIC_AUTH_ACTIVE=true
      - N8N_BASIC_AUTH_USER=${N8N_USER:-admin}
      - N8N_BASIC_AUTH_PASSWORD=${N8N_PASSWORD:-localn8npass}
      - N8N_HOST=${N8N_HOST:-localhost}
      - N8N_PORT=5678
      - N8N_PROTOCOL=http
      - WEBHOOK_URL=http://localhost:5678/
      - GENERIC_TIMEZONE=Europe/Moscow
      - TZ=Europe/Moscow
      # Подключение к PostgreSQL для хранения данных n8n
      - DB_TYPE=postgresdb
      - DB_POSTGRESDB_HOST=postgres
      - DB_POSTGRESDB_PORT=5432
      - DB_POSTGRESDB_DATABASE=n8n
      - DB_POSTGRESDB_USER=${POSTGRES_USER:-neuroguardian}
      - DB_POSTGRESDB_PASSWORD=${POSTGRES_PASSWORD:-localdevpassword}
      # Секреты для шифрования credentials
      - N8N_ENCRYPTION_KEY=${N8N_ENCRYPTION_KEY:-your-32-char-encryption-key-here}
    volumes:
      - n8n_data:/home/node/.n8n
      - ./n8n/workflows:/home/node/workflows:ro
    ports:
      - "5678:5678"
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    healthcheck:
      test: ["CMD-SHELL", "wget --spider -q http://localhost:5678/healthz || exit 1"]
      interval: 30s
      timeout: 10s
      retries: 3
    networks:
      - ng_network

4.3 Prisma схема
prismaDownloadCopy code// packages/database/prisma/schema.prisma

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// =============================================
// Пользователи и аутентификация
// =============================================

model User {
  id            String    @id @default(cuid())
  email         String    @unique
  name          String?
  image         String?
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  // Связи
  stores        Store[]
  settings      UserSettings?
  
  @@map("users")
}

model UserSettings {
  id                    String   @id @default(cuid())
  userId                String   @unique
  user                  User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  
  // Настройки уведомлений
  notifyLowStock        Boolean  @default(true)
  notifyPriceChange     Boolean  @default(true)
  notifyLowMargin       Boolean  @default(true)
  lowStockThreshold     Int      @default(10)
  lowMarginThreshold    Float    @default(15.0)
  
  // Настройки агента
  agentVoice            String   @default("professional") // professional, friendly, expert
  defaultMarketplace    String   @default("wb") // wb, ozon, both
  
  createdAt             DateTime @default(now())
  updatedAt             DateTime @updatedAt
  
  @@map("user_settings")
}

// =============================================
// Магазины и интеграции
// =============================================

model Store {
  id            String      @id @default(cuid())
  userId        String
  user          User        @relation(fields: [userId], references: [id], onDelete: Cascade)
  
  name          String
  marketplace   Marketplace
  isActive      Boolean     @default(true)
  
  // API credentials (зашифрованы)
  apiToken      String?     @db.Text
  clientId      String?
  
  // Статус синхронизации
  lastSyncAt    DateTime?
  syncStatus    SyncStatus  @default(PENDING)
  syncError     String?
  
  // Связи
  products      Product[]
  orders        Order[]
  
  createdAt     DateTime    @default(now())
  updatedAt     DateTime    @updatedAt
  
  @@unique([userId, marketplace])
  @@map("stores")
}

enum Marketplace {
  WB
  OZON
}

enum SyncStatus {
  PENDING
  SYNCING
  SUCCESS
  ERROR
}

// =============================================
// Товары
// =============================================

model Product {
  id                String        @id @default(cuid())
  storeId           String
  store             Store         @relation(fields: [storeId], references: [id], onDelete: Cascade)
  
  // Идентификаторы маркетплейса
  externalId        String        // nmId для WB, product_id для Ozon
  sku               String?
  barcode           String?
  
  // Основная информация
  name              String
  brand             String?
  category          String?
  
  // Цены
  price             Float         // Текущая цена
  priceBeforeDiscount Float?      // Цена до скидки
  costPrice         Float?        // Себестоимость
  minPrice          Float?        // Минимальная цена (защита маржи)
  
  // Комиссии и расчёты
  commissionPercent Float?        // Комиссия МП
  logisticsCost     Float?        // Логистика
  sppPercent        Float?        // СПП (WB)
  
  // Остатки
  stock             Int           @default(0)
  reservedStock     Int           @default(0)
  
  // Медиа
  imageUrl          String?
  images            String[]      // Массив URL
  
  // Статус
  isActive          Boolean       @default(true)
  status            ProductStatus @default(ACTIVE)
  
  // Метрики
  rating            Float?
  reviewsCount      Int           @default(0)
  salesCount        Int           @default(0)
  
  // Связи
  priceHistory      PriceHistory[]
  stockHistory      StockHistory[]
  calculations      PriceCalculation[]
  
  createdAt         DateTime      @default(now())
  updatedAt         DateTime      @updatedAt
  
  @@unique([storeId, externalId])
  @@index([storeId])
  @@index([sku])
  @@map("products")
}

enum ProductStatus {
  ACTIVE
  INACTIVE
  OUT_OF_STOCK
  BLOCKED
}

// =============================================
// История цен
// =============================================

model PriceHistory {
  id          String   @id @default(cuid())
  productId   String
  product     Product  @relation(fields: [productId], references: [id], onDelete: Cascade)
  
  price       Float
  priceType   String   // regular, promo, spp
  source      String   // api_sync, manual, agent
  
  createdAt   DateTime @default(now())
  
  @@index([productId, createdAt])
  @@map("price_history")
}

// =============================================
// История остатков
// =============================================

model StockHistory {
  id          String   @id @default(cuid())
  productId   String
  product     Product  @relation(fields: [productId], references: [id], onDelete: Cascade)
  
  stock       Int
  warehouse   String?
  source      String   // api_sync, manual
  
  createdAt   DateTime @default(now())
  
  @@index([productId, createdAt])
  @@map("stock_history")
}

// =============================================
// Расчёты калькулятора
// =============================================

model PriceCalculation {
  id                    String   @id @default(cuid())
  productId             String
  product               Product  @relation(fields: [productId], references: [id], onDelete: Cascade)
  
  // Входные данные
  inputCostPrice        Float
  inputPrice            Float
  inputLogistics        Float
  inputCommission       Float
  inputSpp              Float
  inputCardDiscount     Float
  inputReturnRate       Float
  
  // Результаты
  calculatedMargin      Float
  calculatedProfit      Float
  recommendedPrice      Float
  breakEvenPrice        Float
  
  // Предупреждения
  warnings              String[] // Массив предупреждений
  
  createdAt             DateTime @default(now())
  
  @@index([productId])
  @@map("price_calculations")
}

// =============================================
// Заказы
// =============================================

model Order {
  id              String      @id @default(cuid())
  storeId         String
  store           Store       @relation(fields: [storeId], references: [id], onDelete: Cascade)
  
  externalId      String      // Номер заказа в МП
  status          OrderStatus
  
  // Финансы
  totalPrice      Float
  commission      Float?
  logistics       Float?
  profit          Float?
  
  // Даты
  orderedAt       DateTime
  deliveredAt     DateTime?
  
  createdAt       DateTime    @default(now())
  updatedAt       DateTime    @updatedAt
  
  @@unique([storeId, externalId])
  @@index([storeId, orderedAt])
  @@map("orders")
}

enum OrderStatus {
  NEW
  CONFIRMED
  ASSEMBLED
  SHIPPED
  DELIVERED
  CANCELLED
  RETURNED
}

// =============================================
// Уведомления и алерты
// =============================================

model Alert {
  id          String      @id @default(cuid())
  storeId     String
  productId   String?
  
  type        AlertType
  severity    AlertSeverity
  title       String
  message     String
  data        Json?       // Дополнительные данные
  
  isRead      Boolean     @default(false)
  isResolved  Boolean     @default(false)
